{{/* Project override of Hextra's article pager. Hextra pages with .PrevInSection / .NextInSection, which are scoped to the parent section. For the DSGVO that would break the chain at every Kapitel and Abschnitt boundary -- Art. 31 would have no "next", Art. 32 no "previous", and Art. 23 (sole article in Kap. III Abschn. 5) neither. Readers expect to walk Art. 1 -> Art. 99 continuously, so for articles we page over a flat list ordered by article number instead. Every other section keeps stock behaviour. The order key is derived from the slug (art-32 -> 32), so it stays correct no matter how the chapter/section tree is rearranged. */}} {{- $reversePagination := .Store.Get "reversePagination" | default false -}} {{- $prev := cond $reversePagination .PrevInSection .NextInSection -}} {{- $next := cond $reversePagination .NextInSection .PrevInSection -}} {{- if and (eq .Section "dsgvo") (hasPrefix .Slug "art-") -}} {{- $ordered := slice -}} {{- range where .Site.RegularPages "Section" "dsgvo" -}} {{- if hasPrefix .Slug "art-" -}} {{- $ordered = $ordered | append (dict "no" (int (strings.TrimPrefix "art-" .Slug)) "page" .) -}} {{- end -}} {{- end -}} {{- $ordered = sort $ordered "no" -}} {{- $here := int (strings.TrimPrefix "art-" .Slug) -}} {{- $prev = false -}} {{- $next = false -}} {{- range $i, $e := $ordered -}} {{- if eq $e.no $here -}} {{- if gt $i 0 -}}{{- $prev = (index $ordered (sub $i 1)).page -}}{{- end -}} {{- if lt $i (sub (len $ordered) 1) -}}{{- $next = (index $ordered (add $i 1)).page -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq .Params.prev false }} {{- if $reversePagination }}{{ $next = false }}{{ else }}{{ $prev = false }}{{ end -}} {{ else }} {{- with .Params.prev -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $next = . }}{{ else }}{{ $prev = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq .Params.next false }} {{- if $reversePagination }}{{ $prev = false }}{{ else }}{{ $next = false }}{{ end -}} {{ else }} {{- with .Params.next -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $prev = . }}{{ else }}{{ $next = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- if $prev -}} {{- $linkTitle := partial "utils/title" $prev -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx:inline hx:h-5 hx:shrink-0 hx:ltr:rotate-180\"") -}} {{- $linkTitle -}} {{- end -}} {{- if $next -}} {{- $linkTitle := partial "utils/title" $next -}} {{- $linkTitle -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx:inline hx:h-5 hx:shrink-0 hx:rtl:-rotate-180\"") -}} {{- end -}}
